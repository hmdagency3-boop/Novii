<script>
// --- SUPABASE CONFIGURATION ---
// IMPORTANT: REPLACE THESE WITH YOUR ACTUAL SUPABASE KEYS
const SUPABASE_URL = 'YOUR_SUPABASE_URL';
const SUPABASE_KEY = 'YOUR_SUPABASE_KEY';

let supabase;

try {
    supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);
} catch (e) {
    console.error("Supabase init failed", e);
}

// --- STATE MANAGEMENT ---
// We keep a local copy for fast rendering, but sync with DB
let state = {
    admins: [],
    agencies: [],
    supporters: []
};
let searchResults = [];
let openTabs = [];
let activeTabId = null;
let currentModalState = { type: null, id: null, mode: 'add' }; 

// --- AUTHENTICATION ---
async function handleLogin() {
    const email = document.getElementById('email').value;
    const password = document.getElementById('password').value;
    const btnText = document.getElementById('login-btn-text');
    const errorMsg = document.getElementById('login-error');

    if(!email || !password) {
        errorMsg.innerText = "Please enter email and password";
        errorMsg.style.display = 'block';
        return;
    }

    btnText.innerText = "Authenticating...";
    
    // Check if keys are set
    if(SUPABASE_URL.includes('YOUR_SUPABASE') || SUPABASE_KEY.includes('YOUR_SUPABASE')) {
        alert("ERROR: Please edit the code to add your Supabase URL and Key in lines 515-516.");
        btnText.innerText = "Config Error";
        return;
    }

    const { data, error } = await supabase.auth.signInWithPassword({
        email: email,
        password: password,
    });

    if (error) {
        errorMsg.innerText = error.message;
        errorMsg.style.display = 'block';
        btnText.innerText = "Secure Login";
    } else {
        document.getElementById('login-page').style.display = 'none';
        document.getElementById('admin-ui').style.display = 'flex';
        // Initial Data Load
        await loadAllData();
        openTab('dashboard', 'Dashboard');
    }
}

async function handleLogout() {
    await supabase.auth.signOut();
    location.reload();
}

// --- DATA FETCHING (SYNC) ---
async function loadAllData() {
    // 1. Fetch Admins
    const { data: admins, error: err1 } = await supabase.from('managed_admins').select('*');
    if(admins) state.admins = admins;

    // 2. Fetch Agencies
    const { data: agencies, error: err2 } = await supabase.from('agencies').select('*');
    if(agencies) state.agencies = agencies;

    // 3. Fetch Supporters
    const { data: supporters, error: err3 } = await supabase.from('supporters').select('*');
    if(supporters) state.supporters = supporters;

    updateCounts();
    // Re-render current tab if active
    if(activeTabId) renderActivePage();
}

function updateCounts() {
    document.getElementById('cnt-agencies').innerText = state.agencies.length;
    document.getElementById('cnt-supporters').innerText = state.supporters.length;
    document.getElementById('cnt-admins').innerText = state.admins.length;
}

// --- TABS SYSTEM (Unchanged) ---
function openTab(id, title, type = 'main', dataId = null) {
  let tabId = id;
  if (id === 'details') tabId = 'view-' + dataId; 
  const existingIndex = openTabs.findIndex(t => t.tabId === tabId);
  if (existingIndex !== -1) {
    activeTabId = tabId;
  } else {
    openTabs.push({ tabId, id, title, type, dataId });
    activeTabId = tabId;
  }
  renderTabs();
  renderActivePage();
}

function closeTab(e, tabId) {
  if(e) e.stopPropagation();
  const index = openTabs.findIndex(t => t.tabId === tabId);
  if (index === -1) return;
  openTabs.splice(index, 1);
  if (activeTabId === tabId) {
    activeTabId = openTabs.length > 0 ? openTabs[openTabs.length - 1].tabId : null;
  }
  renderTabs();
  renderActivePage();
}

function renderTabs() {
  document.getElementById('tabs-container').innerHTML = openTabs.map(tab => `
    <div class="tab ${tab.tabId === activeTabId ? 'active' : ''}" onclick="openTab('${tab.id}', '${tab.title}', '${tab.type}', '${tab.dataId}')">
      <span>${tab.title}</span>
      <span class="close-tab" onclick="closeTab(event, '${tab.tabId}')">&times;</span>
    </div>
  `).join('');
}

function renderActivePage() {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  if (!activeTabId) return;
  const currentTab = openTabs.find(t => t.tabId === activeTabId);
  if (currentTab.id === 'details') {
    document.getElementById('template-details').classList.add('active');
    renderDetails(currentTab.type, currentTab.dataId);
  } else {
    const p = document.getElementById(`template-${currentTab.id}`);
    if (p) p.classList.add('active');
    if(['agencies', 'supporters', 'admins'].includes(currentTab.id)) render(currentTab.id.slice(0, -1)); // remove 's'
    if(currentTab.id === 'users') render('user');
  }
}

// --- RENDER LISTS ---
function render(type) {
  const tbody = document.getElementById(`${type}-body`);
  if (!tbody) return;
  
  let data = type === 'agency' ? state.agencies : type === 'supporter' ? state.supporters : type === 'admin' ? state.admins : searchResults;
  const searchText = (document.getElementById(`search-${type}-text`)?.value || '').toLowerCase();

  // Simple Filter
  if (searchText) {
      data = data.filter(d => {
          if (type === 'agency') return (d.code||'').toLowerCase().includes(searchText) || (d.name||'').toLowerCase().includes(searchText);
          if (type === 'supporter') return (d.sid||'').toLowerCase().includes(searchText);
          if (type === 'admin') return (d.erban_no||'').includes(searchText) || (d.nick||'').toLowerCase().includes(searchText);
          return false;
      });
  }

  tbody.innerHTML = data.map(d => {
    const actions = `<div style="display:flex; gap:5px; justify-content:center">
      <button class="btn btn-outline btn-sm" onclick="openTab('details', 'View', '${type}', '${d.id}')">View</button>
      <button class="btn btn-primary btn-sm" onclick="openEditModal('${type}', '${d.id}')">Edit</button>
      <button class="btn btn-danger btn-sm" onclick="deleteRecord('${type}', '${d.id}')">Del</button>
    </div>`;

    if (type === 'agency') return `<tr><td>${d.code}</td><td>${d.name}</td><td>${d.agent_id}</td><td>${d.region}</td><td>${d.admin_id||'-'}</td><td>${d.join_date}</td><td>${actions}</td></tr>`;
    if (type === 'supporter') return `<tr><td>${d.sid}</td><td>${d.program}</td><td>${d.admin_id||'-'}</td><td>${d.join_date}</td><td>${actions}</td></tr>`;
    if (type === 'admin') return `<tr><td><img src="${d.avatar_url}" style="width:24px;border-radius:4px;vertical-align:middle"> ${d.nick}</td><td>${d.erban_no}</td><td>${d.uid}</td><td>${d.role}</td><td><span class="status-badge status-${d.status}">${d.status}</span></td><td>${actions}</td></tr>`;
    if (type === 'user') return `<tr><td>-</td><td><img src="${d.img}" style="width:24px;border-radius:50%"> ${d.nick}</td><td>${d.id}</td><td>${d.uid}</td><td>${d.country}</td></tr>`;
    return '';
  }).join('');
}

// --- CRUD OPERATIONS (SUPABASE) ---
async function deleteRecord(type, id) {
    if(!confirm('Delete this record permanently from Cloud?')) return;
    const table = type === 'agency' ? 'agencies' : type === 'supporter' ? 'supporters' : 'managed_admins';
    
    const { error } = await supabase.from(table).delete().eq('id', id);
    if(error) alert('Delete failed: ' + error.message);
    else await loadAllData();
}

// --- IMPORT LOGIC ---
async function parseData(type) {
    const inputEl = document.getElementById(`${type}-input`);
    const lines = inputEl.value.trim().split('\n');
    const rowsToInsert = [];

    lines.forEach(line => {
        const p = line.split('\t').map(x => x.trim());
        if(p.length < 3) return;
        
        if (type === 'agency') {
            // Mapping text data to DB columns
            rowsToInsert.push({ 
                code: p[1], agent_id: p[2], name: p[3], region: p[4], 
                join_date: sanitizeDate(p[5]), admin_name: p[6], admin_id: p[7] 
            });
        } else if (type === 'supporter') {
            rowsToInsert.push({
                sid: p[1], program: p[2], admin_name: p[3], 
                admin_id: p[4], join_date: sanitizeDate(p[5])
            });
        }
    });

    if(rowsToInsert.length > 0) {
        const table = type === 'agency' ? 'agencies' : 'supporters';
        const { error } = await supabase.from(table).insert(rowsToInsert);
        if(error) alert("Import Error: " + error.message);
        else {
            alert(`Successfully imported ${rowsToInsert.length} records.`);
            inputEl.value = '';
            loadAllData();
        }
    }
}

async function parseAdminBulk() {
    const inputEl = document.getElementById('admin-bulk-input');
    const lines = inputEl.value.trim().split('\n');
    let insertedCount = 0;

    for (let line of lines) {
        const p = line.split('\t').map(x => x.trim());
        if (p.length < 1) continue;
        const erbanNo = p[0];
        
        // Fetch API
        try {
            const res = await fetch(`https://www.sayyouditto.com/pay/payermax/getInfo?no=${erbanNo}`);
            const j = await res.json();
            if (j.data) {
                const { error } = await supabase.from('managed_admins').insert({
                    erban_no: j.data.erbanNo,
                    uid: j.data.uid,
                    nick: j.data.nick,
                    avatar_url: j.data.avatar,
                    country: j.data.country || 'Global',
                    phone: p[1]||'-',
                    role: p[2]||'Admin',
                    status: parseInt(p[3]||'5')
                });
                if(!error) insertedCount++;
            }
        } catch(e) { console.error(e); }
    }
    alert(`Uploaded ${insertedCount} admins.`);
    inputEl.value = '';
    loadAllData();
}

function sanitizeDate(dateStr) {
    if(!dateStr) return null;
    const match = dateStr.match(/(\d{4}-\d{2}-\d{2})/);
    return match ? match[1] : null;
}

async function saveModal() {
    const { type, id, mode } = currentModalState;
    const btn = document.getElementById('modal-save-btn');
    btn.innerText = "Saving...";

    let payload = {};
    const getVal = (id) => document.getElementById(id).value;

    if(type === 'agency') {
        payload = { name: getVal('m-name'), code: getVal('m-code'), agent_id: getVal('m-agentid'), region: getVal('m-region'), admin_id: getVal('m-adminid'), join_date: getVal('m-date') };
    } else if (type === 'supporter') {
        payload = { sid: getVal('m-sid'), program: getVal('m-prog'), admin_id: getVal('m-adminid'), join_date: getVal('m-date') };
    } else if (type === 'admin') {
        if(mode === 'add') {
             // Fetch extra info first
             try {
                 const res = await fetch(`https://www.sayyouditto.com/pay/payermax/getInfo?no=${getVal('m-id')}`);
                 const j = await res.json();
                 if(!j.data) throw new Error("ID Not Found");
                 payload = { erban_no: j.data.erbanNo, uid: j.data.uid, nick: j.data.nick, avatar_url: j.data.avatar, country: j.data.country, phone: getVal('m-phone'), role: getVal('m-role') };
             } catch(e) { alert(e.message); btn.innerText="Save"; return; }
        } else {
            payload = { phone: getVal('m-phone'), role: getVal('m-role'), status: getVal('m-status') };
        }
    }

    let error;
    if(mode === 'add') {
        const res = await supabase.from(type === 'agency' ? 'agencies' : type === 'supporter' ? 'supporters' : 'managed_admins').insert(payload);
        error = res.error;
    } else {
        const res = await supabase.from(type === 'agency' ? 'agencies' : type === 'supporter' ? 'supporters' : 'managed_admins').update(payload).eq('id', id);
        error = res.error;
    }

    if(error) alert(error.message);
    else {
        closeModal();
        loadAllData();
    }
    btn.innerText = "Save to Cloud";
}

function closeModal() { document.getElementById('general-modal').style.display = 'none'; }
